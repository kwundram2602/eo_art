# eo_art v0.1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Pipeline/Builder-style Python library (`eo_art`) for creating 2D art, animations, and 3D scenes from Earth Observation data (DEMs, multispectral rasters, time series).

**Architecture:** `EOData` wraps an `xr.Dataset` with CRS and resolution metadata; pipeline methods return immutable `RenderStep` objects that chain processing stages; `.render(path)` and `.animate(path)` dispatch on file extension; 3D and mesh export are optional extras guarded by `pytest.importorskip`.

**Tech Stack:** xarray, rioxarray, rasterio, geopandas, matplotlib, numpy, imageio, pyyaml; pyvista + trimesh as optional `[3d]` extras.

```python
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path
from typing import TYPE_CHECKING, Literal

import numpy as np
import xarray as xr

if TYPE_CHECKING:
    from ..render2d.composite import CompositeAccessor
    from ..render2d.result import RenderStep
    from ..render3d.scene import Scene3D


def _detect_kind(ds: xr.Dataset) -> Literal["raster", "timeseries", "dem", "vector"]:
    first_var = next(iter(ds.data_vars.values()))
    if "time" in first_var.dims:
        return "timeseries"
    if first_var.sizes.get("band", 2) == 1 and np.issubdtype(first_var.dtype, np.floating):
        return "dem"
    return "raster"


@dataclass(frozen=True)
class EOData:
    ds: xr.Dataset
    crs: str
    resolution: float
    kind: Literal["raster", "timeseries", "dem", "vector"]

    @classmethod
    def from_xarray(cls, ds: xr.Dataset, crs: str, resolution: float = 1.0) -> EOData:
        return cls(ds=ds, crs=crs, resolution=resolution, kind=_detect_kind(ds))

    @classmethod
    def from_file(cls, path: str | Path) -> EOData:
        from .io import load_file
        return load_file(Path(path))

    @property
    def composite(self) -> CompositeAccessor:
        from ..render2d.composite import CompositeAccessor
        return CompositeAccessor(self)

    def hillshade(self, azimuth: float = 315, altitude: float = 45) -> RenderStep:
        from ..render2d.hillshade import compute_hillshade
        return compute_hillshade(self, azimuth=azimuth, altitude=altitude)

    def scene3d(self, dem: EOData | None = None) -> Scene3D:
        from ..render3d.scene import Scene3D
        return Scene3D(dem=dem if dem is not None else self)
```

```python
from __future__ import annotations

from pathlib import Path

import xarray as xr

from .data import EOData
from .errors import CRSMissingError, EODataLoadError


def load_file(path: str | Path) -> EOData:
    import rioxarray  # noqa: F401 — registers .rio accessor

    path = Path(path)
    try:
        da = xr.open_dataarray(str(path), engine="rasterio")
    except Exception as exc:
        raise EODataLoadError(f"Cannot open {path}: {exc}") from exc

    if da.rio.crs is None:
        raise CRSMissingError(f"{path} has no CRS.")

    crs = da.rio.crs.to_string()
    res_x, _ = da.rio.resolution()
    resolution = abs(float(res_x))

    ds = da.to_dataset(name="data")
    return EOData.from_xarray(ds, crs=crs, resolution=resolution)
```

```python
from __future__ import annotations

from dataclasses import dataclass, field
from pathlib import Path

import numpy as np
import xarray as xr


@dataclass(frozen=True)
class RenderStep:
    """Immutable step in a render pipeline. Each transformation returns a new instance."""

    data: xr.DataArray
    dem: xr.DataArray | None = field(default=None, compare=False)

    def render(self, path: str | Path) -> Path:
        path = Path(path)
        suffix = path.suffix.lower()
        if suffix in {".png", ".jpg", ".jpeg", ".tif", ".tiff"}:
            return _render_image(self.data, path)
        if suffix in {".glb", ".obj", ".ply", ".stl"}:
            from ..export.mesh import export_mesh
            return export_mesh(self, path)
        raise ValueError(f"Unsupported output format: {suffix!r}")

    def animate(self, path: str | Path, fps: int = 4) -> Path:
        from ._animate import render_animation
        return render_animation(self, Path(path), fps=fps)

    def style(
        self,
        palette: str = "viridis",
        blend: str = "replace",
        opacity: float = 1.0,
        preset: str | None = None,
    ) -> RenderStep:
        from .style import apply_style
        return apply_style(self, palette=palette, blend=blend, opacity=opacity, preset=preset)

    def hillshade(
        self,
        azimuth: float = 315,
        altitude: float = 45,
        dem: xr.DataArray | None = None,
    ) -> RenderStep:
        from .hillshade import apply_hillshade_to_step
        return apply_hillshade_to_step(self, azimuth=azimuth, altitude=altitude, dem=dem)

    def blend(self, other: RenderStep, mode: str = "overlay", opacity: float = 0.5) -> RenderStep:
        from .style import blend_steps
        return blend_steps(self, other, mode=mode, opacity=opacity)


def _render_image(data: xr.DataArray, path: Path) -> Path:
    import matplotlib.pyplot as plt

    arr = data.values.astype(np.float32)
    if arr.ndim == 3 and arr.shape[0] in (3, 4):
        img = np.moveaxis(arr, 0, -1)
    elif arr.ndim == 2:
        img = arr
    else:
        raise ValueError(f"Cannot render array with shape {arr.shape}")

    plt.imsave(str(path), np.clip(img, 0.0, 1.0))
    return path
```

```python
from __future__ import annotations

from typing import TYPE_CHECKING

import numpy as np
import xarray as xr

if TYPE_CHECKING:
    from ..core.data import EOData
    from .result import RenderStep


class CompositeAccessor:
    def __init__(self, eo: EOData) -> None:
        self._eo = eo

    def rgb(self, bands: list[int] | None = None) -> RenderStep:
        from .result import RenderStep

        ds = self._eo.ds
        data_var = ds["data"] if "data" in ds else next(iter(ds.data_vars.values()))

        if "band" in data_var.dims:
            n = data_var.sizes["band"]
            idx = (bands or list(range(min(3, n))))[:3]
            arrays = [data_var.isel(band=i).values.astype(np.float32) for i in idx]
        else:
            arrays = [data_var.values.astype(np.float32)] * 3

        stacked = np.stack([_stretch(a) for a in arrays], axis=0)  # (3, H, W)
        return RenderStep(data=xr.DataArray(stacked, dims=("band", "y", "x")))

    def ndvi(self, red: str = "B04", nir: str = "B08") -> RenderStep:
        from .result import RenderStep

        ds = self._eo.ds
        r = ds[red].values.astype(np.float32)
        n = ds[nir].values.astype(np.float32)
        ndvi_vals = (n - r) / (n + r + 1e-8)
        return RenderStep(data=xr.DataArray(ndvi_vals.squeeze(), dims=("y", "x")))


def _stretch(arr: np.ndarray, lo: float = 2.0, hi: float = 98.0) -> np.ndarray:
    vmin = float(np.nanpercentile(arr, lo))
    vmax = float(np.nanpercentile(arr, hi))
    return np.clip((arr - vmin) / (vmax - vmin + 1e-8), 0.0, 1.0)
```

```python
from __future__ import annotations

from typing import TYPE_CHECKING

import numpy as np
import xarray as xr
from matplotlib.colors import LightSource

if TYPE_CHECKING:
    from ..core.data import EOData
    from .result import RenderStep


def compute_hillshade(eo: EOData, azimuth: float = 315, altitude: float = 45) -> RenderStep:
    from .result import RenderStep

    data_var = eo.ds["data"] if "data" in eo.ds else next(iter(eo.ds.data_vars.values()))
    dem_2d = data_var.squeeze().values.astype(np.float64)

    ls = LightSource(azdeg=azimuth, altdeg=altitude)
    shaded = ls.hillshade(dem_2d, vert_exag=1.0).astype(np.float32)

    y = data_var.coords.get("y", np.arange(shaded.shape[-2]))
    x = data_var.coords.get("x", np.arange(shaded.shape[-1]))
    result = xr.DataArray(shaded, dims=("y", "x"), coords={"y": y, "x": x})
    dem_da = xr.DataArray(dem_2d, dims=("y", "x"))

    return RenderStep(data=result, dem=dem_da)


def apply_hillshade_to_step(
    step: RenderStep,
    azimuth: float = 315,
    altitude: float = 45,
    dem: xr.DataArray | None = None,
) -> RenderStep:
    from .result import RenderStep as _RenderStep

    dem_arr = dem if dem is not None else step.dem
    if dem_arr is None:
        raise ValueError("No DEM available. Pass dem= explicitly or use EOData.hillshade().")

    ls = LightSource(azdeg=azimuth, altdeg=altitude)
    shaded = ls.hillshade(dem_arr.values.astype(np.float64), vert_exag=1.0).astype(np.float32)
    return _RenderStep(data=xr.DataArray(shaded, dims=("y", "x")), dem=dem_arr)
```

```python
from __future__ import annotations

from pathlib import Path
from typing import TYPE_CHECKING, Any

import numpy as np
import xarray as xr

if TYPE_CHECKING:
    from .result import RenderStep

_STYLES_DIR = Path(__file__).parent.parent / "styles"


def _load_preset(name: str) -> dict[str, Any]:
    import yaml

    path = _STYLES_DIR / f"{name}.yaml"
    if not path.exists():
        raise FileNotFoundError(f"Style preset {name!r} not found at {path}")
    with path.open() as f:
        return yaml.safe_load(f)


def apply_style(
    step: RenderStep,
    palette: str = "viridis",
    blend: str = "replace",
    opacity: float = 1.0,
    preset: str | None = None,
) -> RenderStep:
    from .result import RenderStep as _RenderStep

    if preset is not None:
        cfg = _load_preset(preset)
        palette = cfg.get("palette", palette)

    data = step.data
    if "band" not in data.dims:
        normalized = _normalize(data.values.astype(np.float32))
        rgb = _apply_cmap(normalized, palette)
        result = xr.DataArray(rgb, dims=("band", "y", "x"))
    else:
        result = data

    return _RenderStep(data=result, dem=step.dem)


def blend_steps(
    base: RenderStep,
    overlay: RenderStep,
    mode: str = "overlay",
    opacity: float = 0.5,
) -> RenderStep:
    from .result import RenderStep as _RenderStep

    a = base.data.values.astype(np.float32)
    b = overlay.data.values.astype(np.float32)

    if mode == "replace":
        blended = a * (1.0 - opacity) + b * opacity
    elif mode == "multiply":
        blended = a * b * opacity + a * (1.0 - opacity)
    elif mode == "overlay":
        soft = np.where(a < 0.5, 2 * a * b, 1 - 2 * (1 - a) * (1 - b))
        blended = a * (1.0 - opacity) + soft * opacity
    else:
        raise ValueError(f"Unknown blend mode: {mode!r}")

    return _RenderStep(data=xr.DataArray(np.clip(blended, 0.0, 1.0), dims=base.data.dims), dem=base.dem)


def _normalize(arr: np.ndarray) -> np.ndarray:
    vmin, vmax = float(np.nanmin(arr)), float(np.nanmax(arr))
    if vmax == vmin:
        return np.zeros_like(arr)
    return (arr - vmin) / (vmax - vmin)


def _apply_cmap(arr: np.ndarray, palette: str) -> np.ndarray:
    import matplotlib.cm as cm

    cmap = cm.get_cmap(palette)
    rgb = cmap(arr)[..., :3].astype(np.float32)  # (H, W, 3), drop alpha
    return np.moveaxis(rgb, -1, 0)  # (3, H, W)
```

```python
from __future__ import annotations

from pathlib import Path
from typing import TYPE_CHECKING

import numpy as np

if TYPE_CHECKING:
    from .result import RenderStep


def render_animation(step: RenderStep, path: Path, fps: int = 4) -> Path:
    import imageio

    if "time" not in step.data.dims:
        raise ValueError("animate() requires a RenderStep with a 'time' dimension.")

    frames = _extract_frames(step.data)
    suffix = path.suffix.lower()

    if suffix == ".gif":
        imageio.mimsave(str(path), frames, duration=1000 // fps, loop=0)
    elif suffix in {".mp4", ".webm", ".avi"}:
        try:
            writer = imageio.get_writer(str(path), fps=fps)
            for frame in frames:
                writer.append_data(frame)
            writer.close()
        except Exception as exc:
            raise RuntimeError(
                f"Cannot write {suffix} — install imageio[ffmpeg]: {exc}"
            ) from exc
    else:
        raise ValueError(f"Unsupported animation format: {suffix!r}")

    return path


def _extract_frames(data: "xr.DataArray") -> list[np.ndarray]:  # type: ignore[name-defined]
    frames = []
    for i in range(data.sizes["time"]):
        arr = data.isel(time=i).values.astype(np.float32)
        if arr.ndim == 3 and arr.shape[0] in (3, 4):
            arr = np.moveaxis(arr, 0, -1)
        frames.append((np.clip(arr, 0.0, 1.0) * 255).astype(np.uint8))
    return frames
```

```python
from __future__ import annotations

from pathlib import Path
from typing import TYPE_CHECKING

import numpy as np

if TYPE_CHECKING:
    from .scene import Scene3D


def render_scene(scene: Scene3D, path: Path) -> Path:
    import pyvista as pv

    dem_var = scene.dem.ds["data"] if "data" in scene.dem.ds else next(iter(scene.dem.ds.data_vars.values()))
    dem_2d = dem_var.squeeze().values.astype(np.float32)
    h, w = dem_2d.shape

    xx, yy = np.meshgrid(np.arange(w, dtype=np.float32), np.arange(h, dtype=np.float32))
    grid = pv.StructuredGrid(xx, yy, dem_2d)
    mesh = grid.extract_surface()

    if scene.texture is not None:
        arr = scene.texture.data.values
        if arr.ndim == 3 and arr.shape[0] == 3:
            rgb = (np.moveaxis(arr, 0, -1) * 255).astype(np.uint8)
            mesh.point_data["RGB"] = rgb.reshape(-1, 3)

    suffix = path.suffix.lower()
    if suffix in {".glb", ".obj", ".ply", ".stl", ".vtk"}:
        mesh.save(str(path))
    else:
        raise ValueError(f"Unsupported 3D format: {suffix!r}")

    return path
```

```python
from pathlib import Path
from typing import TYPE_CHECKING

import numpy as np

if TYPE_CHECKING:
    from ..core.data import EOData
    from ..render2d.result import RenderStep


def export_mesh_from_dem(eo: EOData, path: str | Path) -> Path:
    import trimesh

    path = Path(path)
    dem_var = eo.ds["data"] if "data" in eo.ds else next(iter(eo.ds.data_vars.values()))
    dem_2d = dem_var.squeeze().values.astype(np.float32)
    vertices, faces = _dem_to_mesh(dem_2d)
    trimesh.Trimesh(vertices=vertices, faces=faces, process=False).export(str(path))
    return path


def export_mesh(step: RenderStep, path: Path) -> Path:
    import trimesh

    if step.dem is None:
        raise ValueError("export_mesh requires a RenderStep with a DEM (use EOData.hillshade() first).")
    vertices, faces = _dem_to_mesh(step.dem.values.astype(np.float32))
    trimesh.Trimesh(vertices=vertices, faces=faces, process=False).export(str(path))
    return path


def _dem_to_mesh(dem: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    h, w = dem.shape
    xx, yy = np.meshgrid(np.arange(w, dtype=np.float32), np.arange(h, dtype=np.float32))
    vertices = np.stack([xx.ravel(), yy.ravel(), dem.ravel()], axis=-1)
    faces = []
    for row in range(h - 1):
        for col in range(w - 1):
            i = row * w + col
            faces.extend([[i, i + 1, i + w], [i + 1, i + w + 1, i + w]])
    return vertices, np.array(faces, dtype=np.int64)
```

```python
    return eo.hillshade(azimuth=azimuth, altitude=altitude).style(palette=palette)


def ndvi_map(
    eo: EOData,
    red: str = "B04",
    nir: str = "B08",
    palette: str = "RdYlGn",
) -> RenderStep:
    """NDVI vegetation index rendered with a diverging green-red palette."""
    return eo.composite.ndvi(red=red, nir=nir).style(palette=palette)


def false_color_map(
    eo: EOData,
    bands: list[int] | None = None,
    palette: str | None = None,
) -> RenderStep:
    """False color composite, optionally with a style overlay."""
    step = eo.composite.rgb(bands=bands)
    if palette:
        step = step.style(palette=palette)
    return step
```
